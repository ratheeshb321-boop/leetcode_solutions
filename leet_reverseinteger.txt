7. Reverse Integer

Example 1:
Input: x = 123
Output: 321

Example 2:
Input: x = -123
Output: -321

Example 3:
Input: x = 120
Output: 21
 

Constraints: -231 <= x <= 231 - 1
--------------------------------------------------------------------------------------------------------------
class Solution {
    public int reverse(int x) {
        return solution(x,0);
    }
    public int solution(int x, int rev){
        if(x==0)
            return rev;
        if(rev>Integer.MAX_VALUE/10 || rev<Integer.MIN_VALUE/10)
            return 0;

        int rem = x%10;
        x=x/10;
        rev = rev*10 + rem;

        return solution(x,rev);

    }
}
