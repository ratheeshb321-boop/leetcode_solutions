Given a string s, return the longest palindromic substring in s.

Example 1:

Input: s = "babad"
Output: "bab"
Explanation: "aba" is also a valid answer.
Example 2:

Input: s = "cbbd"
Output: "bb"
 
Constraints:
1 <= s.length <= 1000
s consist of only digits and English letters.
---------------------------------------------------------------------------------------------------------------------
class Solution {
    public String longestPalindrome(String s) {
        int len = s.length();
        int resLen = 0;
        String res = "",finalres="";
        int start =0,end=0;
        for(int i=0;i<len;i++){
            start = i;
            end = i;
            while(start>=0 && end<len && s.charAt(start)==s.charAt(end)){
                res = s.substring(start,end+1);
                if(res.length() > resLen){
                     resLen = res.length();
                     finalres = res;
                }
                start--;
                end++;
                }

            start = i;
            end = i+1;
            while(start>=0 && end<len && s.charAt(start)==s.charAt(end)){
                res = s.substring(start,end+1);
                if(res.length() > resLen){
                     resLen = res.length();
                     finalres = res;
                }
                start--;
                end++;
             }
                
                  
            }
        
        return finalres;
    }
}
